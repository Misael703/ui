# Fase 3 — Major 5.0.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `@misael703/ui` 5.0.0: the kit's core carries no company (neutral brand defaults, `@layer ui`), the deprecated surface is gone, Commerce is an opt-in subpath, and the source is organized one folder per component with per-component CSS partials — all documented in `MIGRATION.md`.

**Architecture:** Every breaking change is its own commit on the `release/5.0.0` branch, ordered risk-first (cascade layer → brand defaults → removals → subpath → structure). The public barrel keeps every non-deprecated name; consumers see three kinds of change: rename `@layer`, call `configureBrand(elalbaDefaults)`, and import Commerce from `@misael703/ui/commerce`. Design-token pinning tests keep reading the CSS *source* through one helper (`readKitCss`) so the CSS split cannot silently change what ships.

**Tech Stack:** React 18/19 peer, TypeScript 5 strict, tsup 8 (esm+cjs, dts, splitting), postcss-import + cssnano, vitest 2 + Testing Library + jsdom, Storybook 8.6 (react-vite), smoke gate (Next 16 real app + Playwright + publint/attw).

**Spec:** `docs/superpowers/specs/2026-09-16-kit-cleanup-design.md` (decisions D1–D10; this plan implements D3, D5, D9, D10 and the "Fase 3" scope) and `docs/superpowers/specs/2026-09-16-kit-cleanup-inventory.md` §I12 (exact `file:line` targets, valid at 4.3.0 `ff55f66`; re-grep before editing because Fase 1 and Fase 2 land first).

- [ ] **Step 4: Rewrite `src/brand.ts`**

```ts
/**
 * Brand and locale defaults used by the kit's formatters and by `<Logo>`.
 *
 * The core assumes no country and no company: the defaults below are neutral
 * (no brand name, USD, en-US). An app configures its brand ONCE at boot; the
 * `@misael703/ui/presets/elalba-defaults` subpath ships a ready-made dict.
 *
 * Country data (regions, phone prefix, tax id validation) is passed by the
 * consumer as props — see `<AddressForm fields={...}>`, `<PhoneInput prefix="+56">`
 * and the `@misael703/ui/cl` helpers.
 *
 * Usage:
 *
 * ```tsx
 * import { configureBrand } from '@misael703/ui';
 *
 * configureBrand({
 *   name: 'Northwind Builders',
 *   currency: 'USD',
 *   locale: 'en-US',
 *   logoBasePath: '/static/brand',
 * });
 * ```
 */

export interface BrandDefaults {
  /** Human-readable brand name. Default `alt` of `<Logo>`; empty until configured. */
  name: string;
  /** Base path of the logo assets. Default: `/assets/logos`. */
  logoBasePath: string;
  /** ISO 4217 code (USD, EUR, CLP, …) for money inputs and formatters. */
  currency: string;
  /** BCP 47 locale for Intl.NumberFormat / DateTimeFormat. */
  locale: string;
}

export const BRAND_DEFAULTS: BrandDefaults = {
  name: '',
  logoBasePath: '/assets/logos',
  currency: 'USD',
  locale: 'en-US',
};

// Lazy singleton: the merge into a concrete dict only happens on the first
// `getBrand()` call, so a tree-shaker can drop this whole module if no
// component in the consumer's bundle calls `getBrand()` or `configureBrand()`.
let _overrides: Partial<BrandDefaults> | null = null;
let _cached: BrandDefaults | null = null;

/**
 * Overrides the kit's global defaults. Call once at app boot (in a client
 * provider under Next App Router). Props still override per call site.
 */
export function configureBrand(overrides: Partial<BrandDefaults>): void {
  _overrides = overrides;
  _cached = null;
}

/** Reads the current defaults. Components use it internally. */
export function getBrand(): BrandDefaults {
  if (!_cached) {
    _cached = _overrides ? { ...BRAND_DEFAULTS, ..._overrides } : BRAND_DEFAULTS;
  }
  return _cached;
}

/** Resets to the neutral defaults. Useful in tests. */
export function resetBrand(): void {
  _overrides = null;
  _cached = null;
}
```

import { readFileSync } from 'node:fs';
import { resolve, dirname } from 'node:path';

export type KitCssFile = 'index' | 'root' | 'typography' | 'fonts' | 'preset';

const FILES: Record<KitCssFile, string> = {
  index: 'src/styles/index.css',
  root: 'src/styles/_root.css',
  typography: 'src/styles/_typography.css',
  fonts: 'src/styles/fonts.css',
  preset: 'src/presets/elalba/styles.css',
};

// Only component partials are inlined. `_root.css` / `_typography.css` stay as
// `@import` lines so the text matches what `index.css` contained before the
// split, and token tests keep reading them through `readKitCss('root')`.
const COMPONENT_IMPORT = /^@import\s+"(\.\/components\/[^"]+)"[^;]*;\s*$/gm;

/** Reads a kit stylesheet from the repo root (vitest's cwd), inlining component partials. */
export function readKitCss(name: KitCssFile): string {
  const path = resolve(process.cwd(), FILES[name]);
  const text = readFileSync(path, 'utf8');
  if (name !== 'index') return text;
  return text.replace(COMPONENT_IMPORT, (_m, rel: string) =>
    readFileSync(resolve(dirname(path), rel), 'utf8'),
  );
}

`scripts/css-diff.mjs`:

```js
// Compares two built stylesheets rule-by-rule (selector + at-rule context →
// declarations), ignoring layer blocks, whitespace and source order.
import { readFileSync } from 'node:fs';
import postcss from 'postcss';

function index(file) {
  const root = postcss.parse(readFileSync(file, 'utf8'));
  const map = new Map();
  root.walkRules((rule) => {
    const ctx = [];
    for (let p = rule.parent; p && p.type !== 'root'; p = p.parent) {
      if (p.type === 'atrule' && p.name !== 'layer') ctx.push(`@${p.name} ${p.params}`);
    }
    const key = `${ctx.reverse().join(' > ')} :: ${rule.selector}`;
    const decls = [];
    rule.walkDecls((d) => decls.push(`${d.prop}:${d.value}${d.important ? '!important' : ''}`));
    // Same selector declared twice: last wins per declaration, like the cascade.
    const prev = map.get(key) ?? new Map();
    for (const d of decls) { const [prop] = d.split(':'); prev.set(prop, d); }
    map.set(key, prev);
  });
  return map;
}

const [a, b] = process.argv.slice(2).map(index);
let diff = 0;
for (const [k, va] of a) {
  const vb = b.get(k);
  if (!vb) { console.log(`- ${k}`); diff++; continue; }
  const sa = [...va.values()].sort().join(';'), sb = [...vb.values()].sort().join(';');
  if (sa !== sb) { console.log(`~ ${k}\n    before: ${sa}\n    after:  ${sb}`); diff++; }
}
for (const k of b.keys()) if (!a.has(k)) { console.log(`+ ${k}`); diff++; }
console.log(diff === 0 ? 'stylesheets are equivalent' : `${diff} differing rules`);
process.exit(diff === 0 ? 0 : 1);
```

'use client';
import * as React from 'react';
import { BRAND_DEFAULTS, configureBrand, getBrand, type BrandDefaults } from './brand';
import { LocaleProvider, type LocaleProviderProps } from './locale/LocaleProvider';

const BrandContext = React.createContext<BrandDefaults | null>(null);

export interface UiProviderProps {
  /** Partial brand; missing keys fall back to the neutral defaults. */
  brand?: Partial<BrandDefaults>;
  /** Partial UI-copy overrides (see `LocaleProvider`). */
  messages?: LocaleProviderProps['messages'];
  /** Also apply `brand` to the module-level singleton so non-React utils (`formatCurrency`) agree with the tree. Default `true`. */
  syncGlobal?: boolean;
  children: React.ReactNode;
}

/** One wrapper for brand + messages. Wrap the app root once. */
export function UiProvider({ brand, messages, syncGlobal = true, children }: UiProviderProps) {
  const value = React.useMemo<BrandDefaults>(() => ({ ...BRAND_DEFAULTS, ...brand }), [brand]);
  if (syncGlobal && brand) configureBrand(brand); // idempotent; runs before children render
  return (
    <BrandContext.Provider value={value}>
      <LocaleProvider messages={messages}>{children}</LocaleProvider>
    </BrandContext.Provider>
  );
}

/** Brand from the nearest `UiProvider`, else the module-level singleton. */
export function useBrand(): BrandDefaults {
  return React.useContext(BrandContext) ?? getBrand();
}
